function fmt(n)  { return "Bs " + Number(n).toFixed(2).replace(/\B(?=(\d{3})+(?!\d))/g,","); }
function hoy()   { return new Date().toISOString().split("T")[0]; }
function uid()   { return Math.random().toString(36).slice(2); }
const lineaVacia = () => ({ id:uid(), nombre:"PLA", tipo:"Estándar", color:"Negro", gramos:100 });
const extraVacio = () => ({ id:uid(), descripcion:"", costo:0 });

function costoFijoPorHoras(horas, cfg) {
  return (cfg.electricidadKwh * cfg.consumoKw + cfg.desgastePorHora) * horas;
}
function costoMateriales(lineas, mats) {
  return (lineas||[]).reduce((s,l) => {
    const m = mats.find(m=>m.id===l.nombre)||mats[0];
    return s + (l.gramos/1000)*m.costo;
  }, 0);
}
function costoExtras(extras) {
  return (extras||[]).reduce((s,e)=>s+(Number(e.costo)||0), 0);
}
function calcDesglose(t, mats, cfg) {
  const cantidad = Math.max(1, t.cantidad||1);
  const cMat     = costoMateriales(t.lineas||[], mats);
  const cFijo    = costoFijoPorHoras(t.horas||0, cfg);
  const cMO      = Number(t.manoObra)||0;
  const cExt     = costoExtras(t.extras);
  const costoTotalProduccion = cMat + cFijo + cMO + cExt;
  const costoPorUnidad       = costoTotalProduccion / cantidad;
  const precioPorUnidad      = costoPorUnidad * (1 + (t.markup||0)/100);
  const precioTotal          = precioPorUnidad * cantidad;
  return { cMat, cFijo, cMO, cExt, costoTotalProduccion, costoPorUnidad, precioPorUnidad, precioTotal, cantidad };
}

// ── Sección con color ──
function Seccion({ tipo, titulo, subtitulo, children, accion }) {
  const s = SEC[tipo]||SEC.trabajo;
  return (
    <div style={{borderRadius:12, border:`1.5px solid ${s.border}`, overflow:"hidden", marginBottom:0}}>
      <div style={{background:s.border, padding:"10px 16px", display:"flex", justifyContent:"space-between", alignItems:"center"}}>
        <div>
          <span style={{color:"#fff", fontWeight:600, fontSize:13}}>{s.icon} {titulo}</span>
          {subtitulo && <span style={{color:"rgba(255,255,255,.75)", fontSize:11, marginLeft:8}}>{subtitulo}</span>}
        </div>
        {accion}
      </div>
      <div style={{background:s.bg, padding:"14px 16px"}}>{children}</div>
    </div>
  );
}

// ── Input controlado estable ──
function Field({ label, children, span2 }) {
  return (
    <div style={span2?{gridColumn:"1/-1"}:{}}>
      {label && <label style={{fontSize:12, color:"#555", display:"block", marginBottom:4, fontWeight:500}}>{label}</label>}
      {children}
    </div>
  );
}

                      <input style={{...inp,padding:"8px 10px",fontSize:15,minWidth:86,width:"100%",textAlign:"right"}} type="number" min={0} value={m.costo} onChange={e=>updateMat(m.id,e.target.value)}/>
                    </div>
                    <span style={{fontSize:11,color:"#aaa",whiteSpace:"nowrap"}}>{fmt(m.costo/1000)}/g</span>
                  </div>
                ))}
              </div>
            </Seccion>
          </div>
        )}

        {/* ══ MODAL PDF ══ */}
        {pdfModal&&(
          <div style={{position:"fixed",inset:0,background:"rgba(0,0,0,.75)",display:"flex",flexDirection:"column",alignItems:"center",justifyContent:"center",zIndex:2000,padding:"16px"}}>
            <div style={{background:"#fff",borderRadius:14,width:"100%",maxWidth:760,maxHeight:"90vh",display:"flex",flexDirection:"column",overflow:"hidden",boxShadow:"0 8px 40px rgba(0,0,0,.4)"}}>
              <div style={{display:"flex",justifyContent:"space-between",alignItems:"center",padding:"12px 16px",borderBottom:"1.5px solid #eee",background:"#1a3a5c"}}>
                <span style={{fontWeight:700,fontSize:15,color:"#fff"}}>🖨 Vista previa del presupuesto</span>
                <div style={{display:"flex",gap:8}}>
                  <a href={pdfModal.url} download={pdfModal.nombre}
                    style={{padding:"6px 14px",background:"#27ae60",color:"#fff",borderRadius:8,fontWeight:700,fontSize:13,textDecoration:"none",display:"flex",alignItems:"center",gap:4}}>
                    ⬇ Descargar HTML
                  </a>
                  <button onClick={()=>{URL.revokeObjectURL(pdfModal.url);setPdfModal(null);}}
                    style={{padding:"6px 12px",borderRadius:8,border:"none",background:"#e74c3c",color:"#fff",cursor:"pointer",fontWeight:700,fontSize:13}}>✕ Cerrar</button>
                </div>
              </div>
              <p style={{margin:0,padding:"8px 16px",fontSize:11,color:"#888",background:"#f8f9fa",borderBottom:"1px solid #eee"}}>
                💡 Para guardar como PDF: descarga el archivo, ábrelo en tu navegador y usa <strong>Archivo → Imprimir → Guardar como PDF</strong>.
              </p>
              <iframe src={pdfModal.url} style={{flex:1,border:"none",minHeight:500}} title="Vista previa PDF"/>
            </div>
          </div>
        )}

        {/* ══ MODAL EDICIÓN ══ */}
        {editModal&&editData&&(
          <div style={{position:"fixed",inset:0,background:"rgba(0,0,0,.6)",display:"flex",alignItems:"flex-start",justifyContent:"center",zIndex:1000,overflowY:"auto",padding:"16px 8px"}}>
            <div style={{background:"#f4f6fa",borderRadius:14,padding:16,width:"100%",maxWidth:700,border:"2px solid #27ae60"}}>
              <div style={{display:"flex",justifyContent:"space-between",alignItems:"center",marginBottom:14}}>
                <p style={{margin:0,fontWeight:700,fontSize:17,color:"#1a3a5c"}}>✏️ Editar cotización</p>
                <button onClick={()=>{setEditModal(false);setEditData(null);}} style={{padding:"5px 12px",borderRadius:8,border:"1px solid #ddd",background:"#fff",cursor:"pointer",color:"#666",fontSize:13}}>✕ Cerrar</button>
              </div>
              <FormCotizacion data={editData} onChange={changeEdit} mats={mats} desglose={dEdit}/>
              <div style={{display:"flex",gap:10,marginTop:14}}>
                <button onClick={guardarEdicion} style={{flex:1,padding:"12px",background:"#27ae60",color:"#fff",border:"none",borderRadius:10,fontWeight:700,cursor:"pointer",fontSize:14}}>💾 Guardar cambios</button>
                <button onClick={()=>{setEditModal(false);setEditData(null);}} style={{padding:"12px 18px",background:"#fff",border:"1.5px solid #ddd",borderRadius:10,cursor:"pointer",color:"#666",fontSize:14}}>Cancelar</button>
              </div>
            </div>
          </div>
        )}
      </div>
    </div>
  );
}
